-- Tabele główne (INTEGER odpowiada typowi NUMBER(38,0) w Oracle)
CREATE TABLE Klub (
    ID_klubu       INTEGER      DEFAULT 1 NOT NULL,
    Nazwa          VARCHAR2(50) NOT NULL,
    Miasto         VARCHAR2(50) NOT NULL,
    Ulica          VARCHAR2(50) NOT NULL,
    Nr_budynku     VARCHAR2(10) NOT NULL,
    Nr_lokalu      VARCHAR2(10),
    Kod_pocztowy   CHAR(6)      NOT NULL,
    NIP            CHAR(10)     NOT NULL,
    REGON          VARCHAR2(14),
    Data_zalozenia DATE,
    CONSTRAINT pk_klub        PRIMARY KEY (ID_klubu),
    CONSTRAINT chk_klub_jeden CHECK (ID_klubu = 1),
    CONSTRAINT uq_klub_nazwa  UNIQUE (Nazwa),
    CONSTRAINT uq_klub_nip    UNIQUE (NIP),
    CONSTRAINT uq_klub_regon  UNIQUE (REGON),
    CONSTRAINT chk_klub_kod   CHECK (REGEXP_LIKE(Kod_pocztowy, '^[0-9]{2}-[0-9]{3}$')),
    CONSTRAINT chk_klub_nip   CHECK (REGEXP_LIKE(NIP, '^[0-9]{10}$')),
    CONSTRAINT chk_klub_regon CHECK (REGEXP_LIKE(REGON, '^([0-9]{9}|[0-9]{14})$'))
);

CREATE TABLE Pracownik (
    ID_pracownika     INTEGER GENERATED BY DEFAULT AS IDENTITY NOT NULL,
    Imie              VARCHAR2(50) NOT NULL,
    Nazwisko          VARCHAR2(50) NOT NULL,
    Stanowisko        VARCHAR2(50) NOT NULL,
    Data_zatrudnienia DATE         NOT NULL,
    Nr_telefonu       CHAR(9),
    ID_klubu          INTEGER      DEFAULT 1 NOT NULL,
    CONSTRAINT pk_pracownik      PRIMARY KEY (ID_pracownika),
    CONSTRAINT fk_pracownik_klub FOREIGN KEY (ID_klubu) REFERENCES Klub (ID_klubu),
    CONSTRAINT chk_pracownik_tel CHECK (REGEXP_LIKE(Nr_telefonu, '^[0-9]{9}$'))
);

CREATE TABLE Trener (
    ID_pracownika    INTEGER       NOT NULL,
    Specjalizacja    VARCHAR2(100) NOT NULL,
    Nr_licencji      CHAR(14)      NOT NULL,
    Waznosc_licencji DATE          NOT NULL,
    Staz_lat         INTEGER,
    CONSTRAINT pk_trener           PRIMARY KEY (ID_pracownika),
    CONSTRAINT fk_trener_pracownik FOREIGN KEY (ID_pracownika) REFERENCES Pracownik (ID_pracownika),
    CONSTRAINT uq_trener_licencja  UNIQUE (Nr_licencji),
    CONSTRAINT chk_trener_staz     CHECK (Staz_lat >= 0)
);

CREATE TABLE Stadion (
    ID_stadionu  INTEGER GENERATED BY DEFAULT AS IDENTITY NOT NULL,
    Nazwa        VARCHAR2(50) NOT NULL,
    Miasto       VARCHAR2(50) NOT NULL,
    Ulica        VARCHAR2(50) NOT NULL,
    Nr_budynku   VARCHAR2(10) NOT NULL,
    Nr_lokalu    VARCHAR2(10),
    Kod_pocztowy CHAR(6)      NOT NULL,
    Typ_obiektu  VARCHAR2(50) NOT NULL,
    Nawierzchnia VARCHAR2(50) NOT NULL,
    ID_klubu     INTEGER      DEFAULT 1 NOT NULL,
    CONSTRAINT pk_stadion      PRIMARY KEY (ID_stadionu),
    CONSTRAINT fk_stadion_klub FOREIGN KEY (ID_klubu) REFERENCES Klub (ID_klubu),
    CONSTRAINT chk_stadion_kod CHECK (REGEXP_LIKE(Kod_pocztowy, '^[0-9]{2}-[0-9]{3}$'))
);

CREATE TABLE Godziny_otwarcia (
    ID_godzin          INTEGER GENERATED BY DEFAULT AS IDENTITY NOT NULL,
    ID_stadionu        INTEGER   NOT NULL,
    Dzien_tygodnia     NUMBER(1) NOT NULL,
    Godzina_otwarcia   CHAR(5)   NOT NULL,
    Godzina_zamkniecia CHAR(5)   NOT NULL,
    CONSTRAINT pk_godziny         PRIMARY KEY (ID_godzin),
    CONSTRAINT fk_godziny_stadion FOREIGN KEY (ID_stadionu) REFERENCES Stadion (ID_stadionu),
    CONSTRAINT uq_godziny_dzien   UNIQUE (ID_stadionu, Dzien_tygodnia),
    CONSTRAINT chk_godziny_dzien  CHECK (Dzien_tygodnia BETWEEN 1 AND 7),
    CONSTRAINT chk_godziny_otw    CHECK (REGEXP_LIKE(Godzina_otwarcia, '^([01][0-9]|2[0-3]):[0-5][0-9]$')),
    CONSTRAINT chk_godziny_zam    CHECK (REGEXP_LIKE(Godzina_zamkniecia, '^([01][0-9]|2[0-3]):[0-5][0-9]$')),
    CONSTRAINT chk_godziny_zakres CHECK (Godzina_zamkniecia > Godzina_otwarcia)
);

CREATE TABLE Cwiczenie (
    ID_cwiczenia     INTEGER GENERATED BY DEFAULT AS IDENTITY NOT NULL,
    Nazwa            VARCHAR2(50) NOT NULL,
    Dyscyplina       VARCHAR2(50) NOT NULL,
    Czas_trwania_min INTEGER      NOT NULL,
    Poziom_trudnosci INTEGER      NOT NULL,
    Liczba_powtorzen INTEGER,
    Opis             CLOB,
    ID_klubu         INTEGER      DEFAULT 1 NOT NULL,
    CONSTRAINT pk_cwiczenie      PRIMARY KEY (ID_cwiczenia),
    CONSTRAINT fk_cwiczenie_klub FOREIGN KEY (ID_klubu) REFERENCES Klub (ID_klubu),
    CONSTRAINT chk_cwic_czas     CHECK (Czas_trwania_min > 0),
    CONSTRAINT chk_cwic_poziom   CHECK (Poziom_trudnosci BETWEEN 1 AND 5),
    CONSTRAINT chk_cwic_powt     CHECK (Liczba_powtorzen > 0)
);

CREATE TABLE Sportowiec (
    ID_sportowca    INTEGER GENERATED BY DEFAULT AS IDENTITY NOT NULL,
    Imie            VARCHAR2(50) NOT NULL,
    Nazwisko        VARCHAR2(50) NOT NULL,
    Data_urodzenia  DATE         NOT NULL,
    Dyscyplina      VARCHAR2(50) NOT NULL,
    Data_dolaczenia DATE         NOT NULL,
    Nr_licencji     CHAR(14),
    CONSTRAINT pk_sportowiec          PRIMARY KEY (ID_sportowca),
    CONSTRAINT uq_sportowiec_licencja UNIQUE (Nr_licencji),
    CONSTRAINT chk_sportowiec_daty    CHECK (Data_dolaczenia >= Data_urodzenia)

CREATE TABLE Przyrzad (
    ID_przyrzadu    INTEGER GENERATED BY DEFAULT AS IDENTITY NOT NULL,
    Rodzaj          VARCHAR2(50) NOT NULL,

-- Wyzwalacze realizujące reguły biznesowe
-- (Uruchamiaj po wyłączeniu "Enable SQL parameters" w ustawieniach DBeaver; bez znaków "/")
CREATE OR REPLACE TRIGGER trg_pracownik_data_zatrudnienia
BEFORE INSERT OR UPDATE ON Pracownik
FOR EACH ROW
BEGIN
    IF :NEW.Data_zatrudnienia > SYSDATE THEN
        RAISE_APPLICATION_ERROR(-20001, 'Data zatrudnienia nie może być późniejsza niż aktualna data.');
    END IF;
END;

CREATE OR REPLACE TRIGGER trg_trener_licencja_wazna
BEFORE INSERT OR UPDATE ON Trener
FOR EACH ROW
BEGIN
    IF :NEW.Waznosc_licencji < TRUNC(SYSDATE) THEN
        RAISE_APPLICATION_ERROR(-20002, 'Trener musi posiadać ważną licencję trenerską.');
    END IF;
END;

CREATE OR REPLACE TRIGGER trg_grafik_walidacja
BEFORE INSERT OR UPDATE ON Grafik
FOR EACH ROW
DECLARE
    v_waznosc     DATE;
    v_stadion_prz INTEGER;
BEGIN
    IF :NEW.ID_pracownika IS NOT NULL THEN
        SELECT MAX(Waznosc_licencji) INTO v_waznosc
        FROM Trener
        WHERE ID_pracownika = :NEW.ID_pracownika;

        IF v_waznosc IS NOT NULL AND v_waznosc < TRUNC(:NEW.Czas_rozpoczecia) THEN
            RAISE_APPLICATION_ERROR(-20003, 'Licencja trenera wygasa przed rozpoczęciem zajęć.');
        END IF;
    END IF;

    IF :NEW.ID_przyrzadu IS NOT NULL THEN
        SELECT MAX(ID_stadionu) INTO v_stadion_prz
        FROM Przyrzad
        WHERE ID_przyrzadu = :NEW.ID_przyrzadu;

        IF v_stadion_prz IS NOT NULL AND v_stadion_prz <> :NEW.ID_stadionu THEN
            RAISE_APPLICATION_ERROR(-20004, 'Wybrany przyrząd znajduje się na innym stadionie.');
        END IF;
    END IF;
END;

CREATE OR REPLACE TRIGGER trg_grafik_kolizje
AFTER INSERT OR UPDATE ON Grafik
DECLARE
    v_kolizje INTEGER;
BEGIN
    -- Brak nakładających się treningów tego samego sportowca
    SELECT COUNT(*) INTO v_kolizje
    FROM Grafik g1
    JOIN Grafik g2
      ON g1.ID_sportowca = g2.ID_sportowca
     AND g1.ID_wpisu < g2.ID_wpisu
     AND g1.Czas_rozpoczecia < g2.Czas_zakonczenia
     AND g2.Czas_rozpoczecia < g1.Czas_zakonczenia;
    IF v_kolizje > 0 THEN
        RAISE_APPLICATION_ERROR(-20005, 'Sportowiec ma już zaplanowany trening w tym czasie.');
    END IF;

    -- Brak nakładających się treningów tego samego trenera
    SELECT COUNT(*) INTO v_kolizje
    FROM Grafik g1
    JOIN Grafik g2
      ON g1.ID_pracownika = g2.ID_pracownika
     AND g1.ID_wpisu < g2.ID_wpisu
     AND g1.Czas_rozpoczecia < g2.Czas_zakonczenia
     AND g2.Czas_rozpoczecia < g1.Czas_zakonczenia;
    IF v_kolizje > 0 THEN
        RAISE_APPLICATION_ERROR(-20006, 'Trener prowadzi już inne zajęcia w tym czasie.');
    END IF;

    -- Brak nakładających się rezerwacji tego samego przyrządu
    SELECT COUNT(*) INTO v_kolizje
    FROM Grafik g1
    JOIN Grafik g2
      ON g1.ID_przyrzadu = g2.ID_przyrzadu
     AND g1.ID_wpisu < g2.ID_wpisu
     AND g1.Czas_rozpoczecia < g2.Czas_zakonczenia
     AND g2.Czas_rozpoczecia < g1.Czas_zakonczenia;
    IF v_kolizje > 0 THEN
        RAISE_APPLICATION_ERROR(-20007, 'Wybrany przyrząd jest już zarezerwowany w tym czasie.');
    END IF;
END;

-- ================================================================================
-- CZĘŚĆ 4: ZAPYTANIA DO ZRZUTÓW EKRANU (SEKCJA 5.4 RAPORTU)
-- ================================================================================

-- ZAPYTANIE 1 (zapisz zrzut ekranu jako: pok1.png)
-- Szczegółowy harmonogram zaplanowanych treningów
SELECT TO_CHAR(g.Czas_rozpoczecia, 'YYYY-MM-DD HH24:MI') AS start_treningu,
       TO_CHAR(g.Czas_zakonczenia, 'HH24:MI') AS koniec,
       c.Nazwa AS cwiczenie,
       st.Nazwa AS stadion,
       sp.Imie || ' ' || sp.Nazwisko AS sportowiec,
       NVL2(p.ID_pracownika, p.Imie || ' ' || p.Nazwisko, 'Trening własny')
           AS trener,
       NVL(pr.Nazwa, 'Brak') AS przyrzad
FROM Grafik g
JOIN Cwiczenie c      ON g.ID_cwiczenia  = c.ID_cwiczenia
JOIN Stadion st       ON g.ID_stadionu   = st.ID_stadionu
JOIN Sportowiec sp    ON g.ID_sportowca  = sp.ID_sportowca
LEFT JOIN Trener t    ON g.ID_pracownika = t.ID_pracownika
LEFT JOIN Pracownik p ON t.ID_pracownika = p.ID_pracownika
LEFT JOIN Przyrzad pr ON g.ID_przyrzadu  = pr.ID_przyrzadu
ORDER BY g.Czas_rozpoczecia;


-- ZAPYTANIE 2 (zapisz zrzut ekranu jako: pok2.png)
-- Wykaz trenerów wraz z liczbą prowadzonych sportowców
SELECT p.Imie, p.Nazwisko, t.Specjalizacja, t.Nr_licencji,
       COUNT(ts.ID_sportowca) AS liczba_sportowcow
FROM Pracownik p
JOIN Trener t ON p.ID_pracownika = t.ID_pracownika
LEFT JOIN Trener_Sportowiec ts ON t.ID_pracownika = ts.ID_pracownika
GROUP BY p.ID_pracownika, p.Imie, p.Nazwisko, t.Specjalizacja, t.Nr_licencji
ORDER BY liczba_sportowcow DESC, p.Nazwisko, p.Imie;


-- ZAPYTANIE 3 (zapisz zrzut ekranu jako: pok3.png)
-- Wyszukiwanie przyrządów wymagających naprawy lub przeglądu
SELECT pr.ID_przyrzadu, pr.Rodzaj, pr.Nazwa,
       pr.Stan_techniczny, s.Nazwa AS stadion
FROM Przyrzad pr
JOIN Stadion s ON pr.ID_stadionu = s.ID_stadionu
WHERE pr.Stan_techniczny IN ('Do naprawy', 'Uszkodzony', 'Wymaga przeglądu')
ORDER BY s.Nazwa, pr.Nazwa;
